cmake_minimum_required(VERSION 3.0.2)
project(uav_camera_stream)
add_compile_options(-std=c++14 -Wall -Wextra -Wpedantic)
find_package(catkin REQUIRED COMPONENTS roscpp diagnostic_msgs std_srvs geometry_msgs sensor_msgs std_msgs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-video-1.0)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS roscpp diagnostic_msgs std_srvs geometry_msgs sensor_msgs std_msgs)
add_library(camera_overlay src/overlay.cpp)
target_include_directories(camera_overlay PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(camera_overlay ${catkin_LIBRARIES})
add_executable(camera_stream_node src/camera_stream_node.cpp)
target_include_directories(camera_stream_node PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(camera_stream_node camera_overlay ${catkin_LIBRARIES} PkgConfig::GSTREAMER)
install(TARGETS camera_stream_node camera_overlay
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  add_executable(overlay_integration_test EXCLUDE_FROM_ALL test/overlay_integration_test.cpp)
  target_link_libraries(overlay_integration_test camera_overlay ${catkin_LIBRARIES})
  catkin_add_gtest(overlay_test test/overlay_test.cpp)
  if(TARGET overlay_test)
    target_link_libraries(overlay_test camera_overlay ${catkin_LIBRARIES})
  endif()
endif()
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
